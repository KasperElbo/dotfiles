\begin{document}
\cssheettitle{macOS (AeroSpace)}{Apple Silicon workstation --- modifier is Control+Option, matching Sway's \$mod}

\begin{multicols}{2}

\cssection{Launch}
\begin{cskeys}{0.34\linewidth}
\csrow{Control+Option+Enter}{Open a new Ghostty window}
\csrow{Command+Space}{Native Spotlight (macOS default, not AeroSpace)}
\end{cskeys}

\cssection{Navigate}
\begin{cskeys}{0.40\linewidth}
\csrow{Control+Option+H/J/K/L}{Focus left/down/up/right, across displays}
\end{cskeys}

\cssection{Move}
\begin{cskeys}{0.40\linewidth}
\csrow{Ctrl+Opt+Shift+H/J/K/L}{Move/reorder left/down/up/right, across displays}
\end{cskeys}

\cssection{Workspaces}
\begin{cskeys}{0.34\linewidth}
\csrow{Control+Option+1..9}{Switch AeroSpace workspace 1-9}
\csrow{Ctrl+Opt+Shift+1..9}{Move window to workspace 1-9}
\csrow{Ctrl+Opt+Cmd+H/J/K/L}{3x3 grid: left/down/up/right, wraps}
\end{cskeys}
\csnote{Same wrapped 1-9 grid as Sway (\texttt{aerospace-workspace-grid}); see the Sway sheet's diagram.}

\cssection{Layout}
\begin{cskeys}{0.34\linewidth}
\csrow{Control+Option+B/V}{Horizontal / vertical tiles}
\csrow{Control+Option+S/W}{Vertical / horizontal accordion}
\csrow{Ctrl+Opt+Shift+Space}{Toggle floating/tiling}
\csrow{Control+Option+F}{Fullscreen within the current Space}
\end{cskeys}

\columnbreak

\cssection{Resize}
\begin{cskeys}{0.34\linewidth}
\csrow{Control+Option+R}{Enter resize mode}
\csrow{H/J/K/L}{Shrink/grow width and height}
\csrow{Enter, Esc}{Exit resize mode}
\end{cskeys}

\cssection{Displays}
\begin{cskeys}{0.34\linewidth}
\csrow{Control+Option+Tab}{Focus next display}
\csrow{Ctrl+Opt+Shift+Tab}{Move window to next display}
\csrow{Ctrl+Opt+Cmd+Tab}{Move workspace to next display}
\end{cskeys}
\csnote{Shift is the window tier and Cmd the workspace tier, as with the numbered workspaces. The cycle wraps; Control+Option+H/J/K/L already crosses display boundaries without it.}

\cssection{System}
\begin{cskeys}{0.34\linewidth}
\csrow{Ctrl+Opt+Shift+C}{Close focused window}
\csrow{Ctrl+Opt+Shift+R}{Reload AeroSpace config}
\end{cskeys}
\csnote{AeroSpace needs only Accessibility permission (System Settings $\to$ Privacy \& Security) and starts at login. SIP and Gatekeeper stay enabled --- see docs/platforms/macos.md for the AeroSpace-vs-yabai decision record.}

\cssection{Clipboard \& screenshots}
\begin{cskeys}{0.34\linewidth}
\csrow{pbcopy / pbpaste}{Pipe to/from the system clipboard}
\csrow{Cmd+Shift+3/4/5}{Full/region/toolbar screenshot (macOS default)}
\end{cskeys}
\csnote{Screenshot destination is managed to \textasciitilde/Pictures/Screenshots by the installer's deliberate defaults; the shortcuts themselves are unmodified macOS defaults.}

\renewcommand{\cstermlegend}{\cslegend{The keys below are Ghostty's own macOS defaults --- Cmd where Linux uses Ctrl+Shift. \texttt{ghostty +list-keybinds --default} prints what the installed version actually has.}}
% Ghostty sets no `keybind =` in this repository, so every row is an unmodified
% upstream default taken from Ghostty's own macOS defaults; each is registered
% in config/actions.tsv as origin=upstream.
\renewcommand{\cstermdefaults}{%
\begin{cskeys}{0.30\linewidth}
\csrow{Cmd+C / Cmd+V}{Copy / paste}
\csrow{Cmd+T / Cmd+W}{New tab / close the focused surface}
\csrow{Cmd+Shift+[ / ]}{Previous / next tab}
\csrow{Cmd+1..8, Cmd+9}{Go to tab 1-8, last tab}
\csrow{Cmd+D / Cmd+Shift+D}{Split right / down}
\csrow{Cmd+[ / Cmd+]}{Previous / next split}
\csrow{Cmd+Opt+arrows}{Focus the split in that direction}
\csrow{Cmd+Shift+Enter}{Zoom the focused split}
\csrow{Cmd+Enter}{Toggle fullscreen}
\csrow{Cmd+Shift+P}{Command palette}
\csrow{Cmd+K / Cmd+A}{Clear the screen / select all}
\csrow{Cmd+Shift+Up / Down}{Jump to the previous / next shell prompt}
\csrow{Cmd+Plus / Minus / 0}{Grow / shrink / reset the font}
\csrow{Cmd+, / Cmd+Shift+,}{Open / reload the configuration}
\end{cskeys}}
% The shared workflow block prints fzf's Alt+C, and Alt is not a key a Mac
% has: Ghostty is configured to send Left Option as Alt so that row works here.

\csnote{Where the shared blocks below say \texttt{Alt}, press \textbf{Left Option}: Ghostty sends it to the shell as Alt/Meta. \textbf{Right Option} is left alone, so it still types the symbols a Danish layout puts behind it.}
