\cssection{Desktop \& keyboard layout}
\begin{cskeys}{0.30\linewidth}
\csrow{Meta+Alt+K}{Switch US/Danish keyboard layout}
\end{cskeys}
\csnote{This is Plasma 6's own default \emph{Switch to Next Keyboard Layout} shortcut, not a dotfiles binding --- it appears once both layouts are added under System Settings $\to$ Keyboard $\to$ Layouts. The tray keyboard-layout indicator shows the active layout.}
\csnote{Everything else about the KDE desktop --- launcher, window shortcuts, screenshots, clipboard history --- is Plasma's stock configuration, not tracked here. See System Settings $\to$ Shortcuts.}

\cssection{Theme}
\begin{cskeys}{0.30\linewidth}
\csrow{theme <f>}{apply Catppuccin flavour to KDE, Ghostty, terminal tools}
\end{cskeys}
\csnote{Changes the Plasma global theme, colour scheme and cursor too. \texttt{--preserve-wallpaper} keeps the current desktop background; the lock screen always follows the flavour.}
